CREATE TABLE nosuchtable (
    revenue_month                  DATE,
    id                             INTEGER
);

CREATE TABLE raw_customers (
    customer_id                    BIGINT PRIMARY KEY,
    external_customer_id           TEXT NOT NULL,
    email                          TEXT NOT NULL,
    first_name                     TEXT,
    last_name                      TEXT,
    signup_ts                      TIMESTAMPTZ NOT NULL,
    acquisition_channel            TEXT,
    country_code                   TEXT,
    marketing_opt_in               BOOLEAN DEFAULT false,
    customer_status                TEXT NOT NULL
);

CREATE TABLE raw_orders (
    order_id                       BIGINT PRIMARY KEY,
    customer_id                    BIGINT NOT NULL REFERENCES raw_customers(customer_id),
    order_number                   TEXT NOT NULL,
    order_ts                       TIMESTAMPTZ NOT NULL,
    order_status                   TEXT NOT NULL,
    currency                       TEXT NOT NULL,
    subtotal_amount                NUMERIC(12, 2) NOT NULL,
    shipping_amount                NUMERIC(12, 2) NOT NULL DEFAULT 0,
    tax_amount                     NUMERIC(12, 2) NOT NULL DEFAULT 0,
    discount_amount                NUMERIC(12, 2) NOT NULL DEFAULT 0,
    coupon_code                    TEXT,
    sales_channel                  TEXT NOT NULL
);

CREATE TABLE raw_order_items (
    order_item_id                  BIGINT PRIMARY KEY,
    order_id                       BIGINT NOT NULL REFERENCES raw_orders(order_id),
    product_id                     BIGINT NOT NULL,
    sku                            TEXT NOT NULL,
    quantity                       INTEGER NOT NULL,
    unit_price_amount              NUMERIC(12, 2) NOT NULL,
    item_discount_amt              NUMERIC(12, 2) NOT NULL DEFAULT 0,
    item_tax_amount                NUMERIC(12, 2) NOT NULL DEFAULT 0,
    fulfillment_status             TEXT NOT NULL
);

CREATE TABLE raw_payments (
    payment_id                     BIGINT PRIMARY KEY,
    order_id                       BIGINT NOT NULL REFERENCES raw_orders(order_id),
    payment_ts                     TIMESTAMPTZ NOT NULL,
    payment_method                 TEXT NOT NULL,
    payment_status                 TEXT NOT NULL,
    payment_type                   TEXT NOT NULL,
    amount                         NUMERIC(12, 2) NOT NULL,
    processor_fee                  NUMERIC(12, 2) NOT NULL DEFAULT 0,
    refunded_amount                NUMERIC(12, 2) NOT NULL DEFAULT 0,
    gateway_txn_id                 TEXT
);

CREATE TABLE raw_products (
    product_id                     BIGINT PRIMARY KEY,
    sku                            TEXT NOT NULL,
    product_name                   TEXT NOT NULL,
    category                       TEXT,
    subcategory                    TEXT,
    brand                          TEXT,
    standard_cost_amt              NUMERIC(12, 2),
    is_active                      BOOLEAN NOT NULL DEFAULT true,
    launched_date                  DATE
);

CREATE VIEW fct_customer_revenue_daily AS
WITH line_revenue_events AS (
         SELECT slip.customer_id,
            slip.order_id,
            slip.order_date AS revenue_date,
            'item_revenue'::text AS revenue_event_type,
            sum(slip.net_line_amount) AS revenue_amount,
            (sum(slip.estimated_line_cost) * ('-1'::integer)::numeric) AS cost_amount,
            sum(slip.estimated_line_margin) AS margin_amount,
            count(DISTINCT slip.product_id) AS distinct_products
           FROM dev.stg_line_items_priced slip
          GROUP BY slip.customer_id, slip.order_id, slip.order_date
        ), refund_events AS (
         SELECT soe.customer_id,
            soe.order_id,
            COALESCE((soe.last_payment_event_ts)::date, soe.order_date) AS revenue_date,
            'refund'::text AS revenue_event_type,
            (sum(soe.refund_amount) * ('-1'::integer)::numeric) AS revenue_amount,
            (0)::numeric(12,2) AS cost_amount,
            (sum(soe.refund_amount) * ('-1'::integer)::numeric) AS margin_amount,
            0 AS distinct_products
           FROM dev.stg_orders_enriched soe
          WHERE (soe.refund_amount > (0)::numeric)
          GROUP BY soe.customer_id, soe.order_id, COALESCE((soe.last_payment_event_ts)::date, soe.order_date)
        ), all_events AS (
         SELECT line_revenue_events.customer_id,
            line_revenue_events.order_id,
            line_revenue_events.revenue_date,
            line_revenue_events.revenue_event_type,
            line_revenue_events.revenue_amount,
            line_revenue_events.cost_amount,
            line_revenue_events.margin_amount,
            line_revenue_events.distinct_products
           FROM line_revenue_events
        UNION ALL
         SELECT refund_events.customer_id,
            refund_events.order_id,
            refund_events.revenue_date,
            refund_events.revenue_event_type,
            refund_events.revenue_amount,
            refund_events.cost_amount,
            refund_events.margin_amount,
            refund_events.distinct_products
           FROM refund_events
        )
 SELECT customer_id,
    revenue_date,
    (date_trunc('month'::text, (revenue_date)::timestamp with time zone))::date AS revenue_month,
    sum(revenue_amount) AS daily_revenue_amount,
    sum(cost_amount) AS daily_cost_amount,
    sum(margin_amount) AS daily_margin_amount,
    sum(distinct_products) AS daily_distinct_products_touched,
    count(DISTINCT order_id) AS daily_order_count,
    count(*) FILTER (WHERE (revenue_event_type = 'refund'::text)) AS refund_event_count,
    sum(revenue_amount) FILTER (WHERE (revenue_event_type = 'refund'::text)) AS refund_revenue_amount,
    (sum(revenue_amount) - abs(sum(cost_amount))) AS contribution_after_cost_amount
   FROM all_events
  GROUP BY customer_id, revenue_date, ((date_trunc('month'::text, (revenue_date)::timestamp with time zone))::date);

CREATE VIEW int_customer_order_metrics AS
WITH customer_orders AS (
         SELECT c.customer_id,
            c.external_customer_id,
            lower(c.email) AS email_normalized,
            TRIM(BOTH FROM ((COALESCE(c.first_name, ''::text) || ' '::text) || COALESCE(c.last_name, ''::text))) AS full_name,
            c.signup_ts,
            (c.signup_ts)::date AS signup_date,
            COALESCE(c.acquisition_channel, 'unknown'::text) AS acquisition_channel,
            COALESCE(c.country_code, 'ZZ'::text) AS country_code,
            c.marketing_opt_in,
            c.customer_status,
            soe.order_id,
            soe.order_date,
            soe.order_month,
            soe.order_status_bucket,
            soe.order_total_amount,
            soe.net_collected_amount,
            soe.refund_amount,
            soe.total_processor_fee,
            soe.total_units,
            soe.sales_channel,
            soe.coupon_code_normalized
           FROM (dev.raw_customers c
             LEFT JOIN dev.stg_orders_enriched soe ON ((soe.customer_id = c.customer_id)))
        )
 SELECT customer_id,
    external_customer_id,
    email_normalized,
    full_name,
    signup_date,
    acquisition_channel,
    country_code,
    marketing_opt_in,
    customer_status,
    min(order_date) AS first_order_date,
    max(order_date) AS last_order_date,
    count(order_id) AS lifetime_order_count,
    count(order_id) FILTER (WHERE (order_status_bucket = 'completed'::text)) AS completed_order_count,
    count(order_id) FILTER (WHERE (order_status_bucket = 'returned'::text)) AS returned_order_count,
    sum(COALESCE(order_total_amount, (0)::numeric)) AS lifetime_order_total_amount,
    sum(COALESCE(net_collected_amount, (0)::numeric)) AS lifetime_net_collected_amount,
    sum(COALESCE(refund_amount, (0)::numeric)) AS lifetime_refund_amount,
    sum(COALESCE(total_processor_fee, (0)::numeric)) AS lifetime_processor_fee_amount,
    sum(COALESCE(total_units, (0)::bigint)) AS lifetime_units,
    (sum(COALESCE(net_collected_amount, (0)::numeric)) / (NULLIF(count(order_id) FILTER (WHERE (order_status_bucket = 'completed'::text)), 0))::numeric) AS avg_completed_order_value,
    (max(
        CASE
            WHEN (coupon_code_normalized IS NOT NULL) THEN 1
            ELSE 0
        END))::boolean AS has_used_coupon,
    (max(
        CASE
            WHEN (sales_channel = 'marketplace'::text) THEN 1
            ELSE 0
        END))::boolean AS has_marketplace_order
   FROM customer_orders
  GROUP BY customer_id, external_customer_id, email_normalized, full_name, signup_date, acquisition_channel, country_code, marketing_opt_in, customer_status;

CREATE VIEW mart_customer_ltv_segments AS
WITH revenue_rollup AS (
         SELECT fcrd.customer_id,
            min(fcrd.revenue_date) AS first_revenue_date,
            max(fcrd.revenue_date) AS last_revenue_date,
            sum(fcrd.daily_revenue_amount) AS total_revenue_amount,
            sum(fcrd.daily_cost_amount) AS total_cost_amount,
            sum(fcrd.daily_margin_amount) AS total_margin_amount,
            sum(fcrd.daily_order_count) AS active_revenue_days_with_orders,
            sum(fcrd.refund_event_count) AS total_refund_events,
            sum(COALESCE(fcrd.refund_revenue_amount, (0)::numeric)) AS total_refund_revenue_amount
           FROM dev.fct_customer_revenue_daily fcrd
          GROUP BY fcrd.customer_id
        ), scored_customers AS (
         SELECT icom.customer_id,
            icom.external_customer_id,
            icom.email_normalized,
            icom.full_name,
            icom.signup_date,
            icom.acquisition_channel,
            icom.country_code,
            icom.marketing_opt_in,
            icom.customer_status,
            icom.first_order_date,
            icom.last_order_date,
            icom.lifetime_order_count,
            icom.completed_order_count,
            icom.returned_order_count,
            icom.lifetime_net_collected_amount,
            icom.avg_completed_order_value,
            icom.has_used_coupon,
            icom.has_marketplace_order,
            COALESCE(rr.total_revenue_amount, (0)::numeric) AS total_revenue_amount,
            COALESCE(rr.total_cost_amount, (0)::numeric) AS total_cost_amount,
            COALESCE(rr.total_margin_amount, (0)::numeric) AS total_margin_amount,
            COALESCE(rr.total_refund_events, (0)::numeric) AS total_refund_events,
            COALESCE(rr.total_refund_revenue_amount, (0)::numeric) AS total_refund_revenue_amount,
            rr.first_revenue_date,
            rr.last_revenue_date,
                CASE
                    WHEN (icom.first_order_date IS NULL) THEN NULL::integer
                    ELSE (CURRENT_DATE - icom.first_order_date)
                END AS customer_age_days,
            (COALESCE(rr.total_revenue_amount, (0)::numeric) - COALESCE(icom.lifetime_processor_fee_amount, (0)::numeric)) AS ltv_after_fees_amount
           FROM (dev.int_customer_order_metrics icom
             LEFT JOIN revenue_rollup rr ON ((rr.customer_id = icom.customer_id)))
        )
 SELECT customer_id,
    external_customer_id,
    email_normalized,
    full_name,
    signup_date,
    acquisition_channel,
    country_code,
    marketing_opt_in,
    customer_status,
    first_order_date,
    last_order_date,
    first_revenue_date,
    last_revenue_date,
    customer_age_days,
    lifetime_order_count,
    completed_order_count,
    returned_order_count,
    total_revenue_amount,
    total_cost_amount,
    total_margin_amount,
    ltv_after_fees_amount,
    avg_completed_order_value,
    total_refund_events,
    total_refund_revenue_amount,
    has_used_coupon,
    has_marketplace_order,
    dense_rank() OVER (ORDER BY total_revenue_amount DESC) AS revenue_rank,
    ntile(5) OVER (ORDER BY total_revenue_amount DESC) AS revenue_quintile,
        CASE
            WHEN (lifetime_order_count = 0) THEN 'prospect'::text
            WHEN ((total_revenue_amount >= (1000)::numeric) AND (completed_order_count >= 5)) THEN 'vip'::text
            WHEN (total_refund_events >= (3)::numeric) THEN 'refund_risk'::text
            WHEN (customer_age_days <= 30) THEN 'new_customer'::text
            WHEN marketing_opt_in THEN 'marketable'::text
            ELSE 'standard'::text
        END AS customer_segment,
        CASE
            WHEN (total_margin_amount < (0)::numeric) THEN 'negative_margin'::text
            WHEN ((total_margin_amount / NULLIF(total_revenue_amount, (0)::numeric)) >= 0.5) THEN 'high_margin'::text
            WHEN ((total_margin_amount / NULLIF(total_revenue_amount, (0)::numeric)) >= 0.2) THEN 'medium_margin'::text
            ELSE 'low_margin'::text
        END AS margin_segment
   FROM scored_customers;

CREATE VIEW mart_segment_health_snapshot AS
WITH recent_revenue AS (
         SELECT fcrd.customer_id,
            sum(fcrd.daily_revenue_amount) FILTER (WHERE (fcrd.revenue_date >= (CURRENT_DATE - '30 days'::interval))) AS revenue_30d_amount,
            sum(fcrd.daily_margin_amount) FILTER (WHERE (fcrd.revenue_date >= (CURRENT_DATE - '30 days'::interval))) AS margin_30d_amount,
            sum(fcrd.daily_order_count) FILTER (WHERE (fcrd.revenue_date >= (CURRENT_DATE - '30 days'::interval))) AS order_count_30d,
            max(fcrd.revenue_date) AS most_recent_revenue_date
           FROM dev.fct_customer_revenue_daily fcrd
          GROUP BY fcrd.customer_id
        ), cohort_context AS (
         SELECT rgc.signup_month,
            rgc.acquisition_channel,
            rgc.country_code,
            rgc.conversion_rate,
            rgc.vip_rate,
            rgc.negative_margin_rate,
            rgc.revenue_per_customer,
            rgc.cohort_margin_rate,
            rgc.cohort_ltv_rank_in_month
           FROM dev.rpt_customer_growth_cohorts rgc
        )
 SELECT mcls.customer_id,
    mcls.external_customer_id,
    mcls.email_normalized,
    (date_trunc('month'::text, (mcls.signup_date)::timestamp with time zone))::date AS signup_month,
    mcls.acquisition_channel,
    mcls.country_code,
    mcls.customer_segment,
    mcls.margin_segment,
    mcls.revenue_quintile,
    mcls.revenue_rank,
    mcls.total_revenue_amount,
    mcls.total_margin_amount,
    mcls.ltv_after_fees_amount,
    mcls.total_refund_events,
    mcls.completed_order_count,
    COALESCE(rr.revenue_30d_amount, (0)::numeric) AS revenue_30d_amount,
    COALESCE(rr.margin_30d_amount, (0)::numeric) AS margin_30d_amount,
    COALESCE(rr.order_count_30d, (0)::numeric) AS order_count_30d,
    rr.most_recent_revenue_date,
    cc.conversion_rate AS cohort_conversion_rate,
    cc.vip_rate AS cohort_vip_rate,
    cc.negative_margin_rate AS cohort_negative_margin_rate,
    cc.revenue_per_customer AS cohort_revenue_per_customer,
    cc.cohort_margin_rate,
    cc.cohort_ltv_rank_in_month,
        CASE
            WHEN ((mcls.customer_segment = 'vip'::text) AND (COALESCE(rr.revenue_30d_amount, (0)::numeric) > (0)::numeric) AND (mcls.margin_segment <> 'negative_margin'::text)) THEN 'healthy_vip'::text
            WHEN ((mcls.total_refund_events >= (3)::numeric) OR (cc.negative_margin_rate >= 0.25)) THEN 'at_risk'::text
            WHEN ((COALESCE(rr.order_count_30d, (0)::numeric) = (0)::numeric) AND (mcls.customer_age_days > 90)) THEN 'dormant'::text
            ELSE 'monitor'::text
        END AS health_status,
    (((((COALESCE(mcls.ltv_after_fees_amount, (0)::numeric) * 0.50) + (COALESCE(rr.revenue_30d_amount, (0)::numeric) * 0.30)) + ((COALESCE(cc.conversion_rate, (0)::numeric) * (100)::numeric) * 0.10)) - (COALESCE(mcls.total_refund_events, (0)::numeric) * (10)::numeric)) - ((COALESCE(cc.negative_margin_rate, (0)::numeric) * (100)::numeric) * 0.10)) AS health_score
   FROM ((dev.mart_customer_ltv_segments mcls
     LEFT JOIN recent_revenue rr ON ((rr.customer_id = mcls.customer_id)))
     LEFT JOIN cohort_context cc ON (((cc.signup_month = (date_trunc('month'::text, (mcls.signup_date)::timestamp with time zone))::date) AND (cc.acquisition_channel = mcls.acquisition_channel) AND (cc.country_code = mcls.country_code))));

CREATE VIEW rpt_customer_growth_cohorts AS
WITH cohort_base AS (
         SELECT mcls.customer_id,
            mcls.signup_date,
            (date_trunc('month'::text, (mcls.signup_date)::timestamp with time zone))::date AS signup_month,
            mcls.acquisition_channel,
            mcls.country_code,
            mcls.customer_segment,
            mcls.margin_segment,
            mcls.completed_order_count,
            mcls.returned_order_count,
            mcls.total_revenue_amount,
            mcls.total_margin_amount,
            mcls.ltv_after_fees_amount,
            mcls.customer_age_days,
                CASE
                    WHEN (mcls.first_order_date IS NULL) THEN 'never_ordered'::text
                    WHEN (mcls.first_order_date <= (mcls.signup_date + '7 days'::interval)) THEN 'converted_week_1'::text
                    WHEN (mcls.first_order_date <= (mcls.signup_date + '30 days'::interval)) THEN 'converted_month_1'::text
                    ELSE 'converted_later'::text
                END AS conversion_speed_bucket
           FROM dev.mart_customer_ltv_segments mcls
        ), cohort_rollup AS (
         SELECT cohort_base.signup_month,
            cohort_base.acquisition_channel,
            cohort_base.country_code,
            count(*) AS cohort_customer_count,
            count(*) FILTER (WHERE (cohort_base.conversion_speed_bucket <> 'never_ordered'::text)) AS converted_customer_count,
            count(*) FILTER (WHERE (cohort_base.customer_segment = 'vip'::text)) AS vip_customer_count,
            count(*) FILTER (WHERE (cohort_base.margin_segment = 'negative_margin'::text)) AS negative_margin_customer_count,
            sum(cohort_base.total_revenue_amount) AS cohort_revenue_amount,
            sum(cohort_base.total_margin_amount) AS cohort_margin_amount,
            sum(cohort_base.ltv_after_fees_amount) AS cohort_ltv_after_fees_amount,
            sum(cohort_base.completed_order_count) AS cohort_completed_orders,
            sum(cohort_base.returned_order_count) AS cohort_returned_orders,
            avg(cohort_base.customer_age_days) AS avg_customer_age_days
           FROM cohort_base
          GROUP BY cohort_base.signup_month, cohort_base.acquisition_channel, cohort_base.country_code
        )
 SELECT signup_month,
    acquisition_channel,
    country_code,
    cohort_customer_count,
    converted_customer_count,
    ((converted_customer_count)::numeric / (NULLIF(cohort_customer_count, 0))::numeric) AS conversion_rate,
    ((vip_customer_count)::numeric / (NULLIF(cohort_customer_count, 0))::numeric) AS vip_rate,
    ((negative_margin_customer_count)::numeric / (NULLIF(cohort_customer_count, 0))::numeric) AS negative_margin_rate,
    cohort_revenue_amount,
    cohort_margin_amount,
    cohort_ltv_after_fees_amount,
    (cohort_revenue_amount / (NULLIF(cohort_customer_count, 0))::numeric) AS revenue_per_customer,
    (cohort_ltv_after_fees_amount / (NULLIF(cohort_customer_count, 0))::numeric) AS ltv_after_fees_per_customer,
    (cohort_margin_amount / NULLIF(cohort_revenue_amount, (0)::numeric)) AS cohort_margin_rate,
    cohort_completed_orders,
    cohort_returned_orders,
    (cohort_returned_orders / NULLIF(cohort_completed_orders, (0)::numeric)) AS cohort_return_rate,
    avg_customer_age_days,
    (cohort_revenue_amount - lag(cohort_revenue_amount) OVER (PARTITION BY acquisition_channel, country_code ORDER BY signup_month)) AS revenue_delta_vs_prior_cohort,
    rank() OVER (PARTITION BY signup_month ORDER BY cohort_ltv_after_fees_amount DESC) AS cohort_ltv_rank_in_month
   FROM cohort_rollup;

CREATE VIEW rpt_executive_revenue_dashboard AS
WITH segment_rollup AS (
         SELECT mshs.signup_month,
            mshs.acquisition_channel,
            mshs.country_code,
            mshs.customer_segment,
            mshs.health_status,
            count(*) AS customer_count,
            sum(mshs.total_revenue_amount) AS segment_total_revenue_amount,
            sum(mshs.total_margin_amount) AS segment_total_margin_amount,
            sum(mshs.ltv_after_fees_amount) AS segment_ltv_after_fees_amount,
            sum(mshs.revenue_30d_amount) AS segment_revenue_30d_amount,
            avg(mshs.health_score) AS avg_health_score,
            count(*) FILTER (WHERE (mshs.health_status = 'at_risk'::text)) AS at_risk_customer_count,
            count(*) FILTER (WHERE (mshs.health_status = 'healthy_vip'::text)) AS healthy_vip_customer_count
           FROM dev.mart_segment_health_snapshot mshs
          GROUP BY mshs.signup_month, mshs.acquisition_channel, mshs.country_code, mshs.customer_segment, mshs.health_status
        ), fact_monthly AS (
         SELECT fcrd.revenue_month,
            sum(fcrd.daily_revenue_amount) AS platform_revenue_amount,
            sum(fcrd.daily_margin_amount) AS platform_margin_amount,
            sum(fcrd.daily_order_count) AS platform_order_count,
            sum(fcrd.refund_event_count) AS platform_refund_event_count
           FROM dev.fct_customer_revenue_daily fcrd
          GROUP BY fcrd.revenue_month
        )
 SELECT sr.signup_month,
    sr.acquisition_channel,
    sr.country_code,
    sr.customer_segment,
    sr.health_status,
    sr.customer_count,
    sr.segment_total_revenue_amount,
    sr.segment_total_margin_amount,
    sr.segment_ltv_after_fees_amount,
    sr.segment_revenue_30d_amount,
    sr.avg_health_score,
    sr.at_risk_customer_count,
    sr.healthy_vip_customer_count,
    fm.platform_revenue_amount,
    fm.platform_margin_amount,
    fm.platform_order_count,
    fm.platform_refund_event_count,
    (sr.segment_total_revenue_amount / NULLIF(fm.platform_revenue_amount, (0)::numeric)) AS segment_revenue_share,
    (sr.segment_total_margin_amount / NULLIF(sr.segment_total_revenue_amount, (0)::numeric)) AS segment_margin_rate,
    ((sr.at_risk_customer_count)::numeric / (NULLIF(sr.customer_count, 0))::numeric) AS segment_at_risk_rate,
    nst.id,
        CASE
            WHEN ((sr.avg_health_score >= (750)::numeric) AND (((sr.at_risk_customer_count)::numeric / (NULLIF(sr.customer_count, 0))::numeric) < 0.10)) THEN 'invest'::text
            WHEN (((sr.segment_total_margin_amount / NULLIF(sr.segment_total_revenue_amount, (0)::numeric)) < (0)::numeric) OR (((sr.at_risk_customer_count)::numeric / (NULLIF(sr.customer_count, 0))::numeric) >= 0.30)) THEN 'intervene'::text
            WHEN (sr.segment_revenue_30d_amount = (0)::numeric) THEN 'watch'::text
            ELSE 'maintain'::text
        END AS executive_action,
    dense_rank() OVER (PARTITION BY sr.signup_month ORDER BY sr.segment_total_revenue_amount DESC) AS segment_revenue_rank,
    sum(sr.segment_total_revenue_amount) OVER (PARTITION BY sr.signup_month ORDER BY sr.segment_total_revenue_amount DESC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS cumulative_revenue_by_month_rank
   FROM ((segment_rollup sr
     LEFT JOIN fact_monthly fm ON ((fm.revenue_month = sr.signup_month)))
     LEFT JOIN dev.nosuchtable nst ON ((fm.revenue_month = nst.revenue_month)));

CREATE VIEW stg_line_items_priced AS
SELECT oi.order_item_id,
    oi.order_id,
    o.customer_id,
    o.order_ts,
    (o.order_ts)::date AS order_date,
    oi.product_id,
    COALESCE(p.sku, oi.sku) AS resolved_sku,
    p.product_name,
    COALESCE(p.category, 'uncategorized'::text) AS category,
    COALESCE(p.subcategory, 'unknown'::text) AS subcategory,
    ((COALESCE(p.category, 'uncategorized'::text) || ' > '::text) || COALESCE(p.subcategory, 'unknown'::text)) AS category_path,
    p.brand,
    oi.quantity,
    oi.unit_price_amount,
    oi.item_discount_amt,
    oi.item_tax_amount,
    ((oi.quantity)::numeric * oi.unit_price_amount) AS gross_line_amount,
    (((oi.quantity)::numeric * oi.unit_price_amount) - COALESCE(oi.item_discount_amt, (0)::numeric)) AS net_line_amount,
    ((((oi.quantity)::numeric * oi.unit_price_amount) - COALESCE(oi.item_discount_amt, (0)::numeric)) + COALESCE(oi.item_tax_amount, (0)::numeric)) AS net_line_amount_with_tax,
    (COALESCE(p.standard_cost_amt, (0)::numeric) * (oi.quantity)::numeric) AS estimated_line_cost,
    ((((oi.quantity)::numeric * oi.unit_price_amount) - COALESCE(oi.item_discount_amt, (0)::numeric)) - (COALESCE(p.standard_cost_amt, (0)::numeric) * (oi.quantity)::numeric)) AS estimated_line_margin,
        CASE
            WHEN (oi.fulfillment_status = ANY (ARRAY['shipped'::text, 'delivered'::text])) THEN true
            ELSE false
        END AS is_fulfilled,
        CASE
            WHEN (p.is_active IS FALSE) THEN 'inactive_product'::text
            WHEN (p.launched_date > (o.order_ts)::date) THEN 'prelaunch_sale'::text
            ELSE 'normal'::text
        END AS product_sale_flag
   FROM ((dev.raw_order_items oi
     JOIN dev.raw_orders o ON ((o.order_id = oi.order_id)))
     LEFT JOIN dev.raw_products p ON ((p.product_id = oi.product_id)));

CREATE VIEW stg_orders_enriched AS
WITH payment_rollup AS (
         SELECT p.order_id,
            min(p.payment_ts) FILTER (WHERE (p.payment_type = 'charge'::text)) AS first_charge_ts,
            max(p.payment_ts) AS last_payment_event_ts,
            sum(
                CASE
                    WHEN (p.payment_type = 'charge'::text) THEN p.amount
                    ELSE (0)::numeric
                END) AS charged_amount,
            sum(
                CASE
                    WHEN (p.payment_type = 'refund'::text) THEN p.amount
                    ELSE (0)::numeric
                END) AS refund_amount,
            sum(p.processor_fee) AS total_processor_fee,
            count(*) FILTER (WHERE (p.payment_status = 'failed'::text)) AS failed_payment_events
           FROM dev.raw_payments p
          GROUP BY p.order_id
        ), item_rollup AS (
         SELECT oi.order_id,
            count(*) AS item_row_count,
            sum(oi.quantity) AS total_units,
            sum(((oi.quantity)::numeric * oi.unit_price_amount)) AS item_gross_amount,
            sum(oi.item_discount_amt) AS item_discount_amount,
            sum(oi.item_tax_amount) AS item_tax_amount
           FROM dev.raw_order_items oi
          GROUP BY oi.order_id
        )
 SELECT (o.order_id)::text AS order_key,
    o.order_id,
    o.customer_id,
    o.order_number,
    o.order_ts,
    (o.order_ts)::date AS order_date,
    (date_trunc('month'::text, o.order_ts))::date AS order_month,
    lower(TRIM(BOTH FROM o.order_status)) AS raw_order_status_normalized,
        CASE
            WHEN (lower(o.order_status) = ANY (ARRAY['paid'::text, 'complete'::text, 'completed'::text, 'shipped'::text])) THEN 'completed'::text
            WHEN (lower(o.order_status) = ANY (ARRAY['cancelled'::text, 'canceled'::text, 'void'::text])) THEN 'cancelled'::text
            WHEN (lower(o.order_status) = ANY (ARRAY['returned'::text, 'refunded'::text])) THEN 'returned'::text
            ELSE 'open'::text
        END AS order_status_bucket,
    o.currency,
    (((COALESCE(o.subtotal_amount, (0)::numeric) + COALESCE(o.shipping_amount, (0)::numeric)) + COALESCE(o.tax_amount, (0)::numeric)) - COALESCE(o.discount_amount, (0)::numeric)) AS order_total_amount,
    COALESCE(ir.item_row_count, (0)::bigint) AS item_row_count,
    COALESCE(ir.total_units, (0)::bigint) AS total_units,
    COALESCE(ir.item_gross_amount, (0)::numeric) AS item_gross_amount,
    COALESCE(ir.item_discount_amount, (0)::numeric) AS item_discount_amount,
    COALESCE(pr.charged_amount, (0)::numeric) AS charged_amount,
    COALESCE(pr.refund_amount, (0)::numeric) AS refund_amount,
    COALESCE(pr.total_processor_fee, (0)::numeric) AS total_processor_fee,
    (COALESCE(pr.charged_amount, (0)::numeric) - COALESCE(pr.refund_amount, (0)::numeric)) AS net_collected_amount,
    pr.first_charge_ts,
    pr.last_payment_event_ts,
    COALESCE(pr.failed_payment_events, (0)::bigint) AS failed_payment_events,
    (COALESCE(pr.charged_amount, (0)::numeric) >= (((COALESCE(o.subtotal_amount, (0)::numeric) + COALESCE(o.shipping_amount, (0)::numeric)) + COALESCE(o.tax_amount, (0)::numeric)) - COALESCE(o.discount_amount, (0)::numeric))) AS is_fully_paid,
    o.sales_channel,
    NULLIF(upper(TRIM(BOTH FROM o.coupon_code)), ''::text) AS coupon_code_normalized
   FROM ((dev.raw_orders o
     LEFT JOIN payment_rollup pr ON ((pr.order_id = o.order_id)))
     LEFT JOIN item_rollup ir ON ((ir.order_id = o.order_id)));
